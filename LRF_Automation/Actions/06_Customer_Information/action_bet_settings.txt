select product in bet setting
    [Arguments]    ${product}
    [Documentation]    select product in bet setting | Number Game
    Select Window
    Select Frame    main
    Select Frame    xpath=//iframe[contains(@src,'BetSetting')]

    Unselect Frame
    wait page load
