select sport and league by index on results
    [Arguments]    ${sport index}    ${league index}
    [Documentation]    This action is used to select Sport and League in Results page based on the index of the values.
    ...
    ...    Example: select 1st Sport and 2nd League
    ...
    ...    select sport and league by index on results | 1 | 2
    Select Window
    Select Frame    main
    Run Keyword If    '${sport index}'!='None'    Select From List By Index    ${sport_listbox_results}    ${sport index}
    wait page load
    Select Window
    Select Frame    main
    Run Keyword If    '${league index}'!='None'    Select From List By Index    ${league_listbox_results}    ${league index}
    wait page load

click result link of tennis match
    [Arguments]    ${from row}    ${to row}
    [Documentation]    This action is used to click result link of tennis match in Result report.
    ...
    ...    Example:
    ...
    ...    click result link of tennis match | 1 | 50
    Select Window
    Select Frame    main
    Set Selenium Speed    0.2
    # Find link in the range of columns. If existed, click on it
    :FOR    ${ac_index}    IN RANGE    ${from row}    ${to row}+1
    \    ${ac_link_status}=    Run Keyword And Return Status    Element Should Be Visible    xpath=//table[@class='fn-rs ']/tbody/tr[${ac_index}]/td[contains(@class,'cl-status')]/div/span[contains(@class,'tennis-detail-link')]
    \    Run Keyword If    '${ac_link_status}'=='True'    Click Element    xpath=//table[@class='fn-rs ']/tbody/tr[${ac_index}]/td[contains(@class,'cl-status')]/div/span[contains(@class,'tennis-detail-link')]
    \    Run Keyword If    '${ac_link_status}'=='True'    Set Selenium Speed    ${cf_speed}
    \    Exit For Loop If    '${ac_link_status}'=='True'
    \    Run Keyword If    '${ac_index}'=='${to row}' and '${ac_link_status}'!='True'    Set Selenium Speed    ${cf_speed}
    \    Run Keyword If    '${ac_index}'=='${to row}' and '${ac_link_status}'!='True'    Pass Execution    There is no link at this moment.
    Unselect Frame
