select product in bet setting
    [Arguments]    ${product}
    [Documentation]    select product in bet setting | Number Game
    Select Window
    Select Frame    main
    Select Frame    xpath=//iframe[contains(@src,'BetSetting')]
    Click Element    ${product_listbox_bet_setting}
    Wait Until Element Is Visible    xpath=//ul/li/a[contains(text(),'${product}')]    5s
    Click Element    xpath=//ul/li/a[contains(text(),'${product}')]
    Unselect Frame
    wait page load
