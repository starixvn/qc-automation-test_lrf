submit filters for daily bet list
    [Arguments]    ${date}    ${currency}    ${based}    ${product}    ${status}
    [Documentation]    This action is used to submit filters for Daily Bet List report.
    ...
    ...    Example:
    ...
    ...    submit filters for daily bet list | Dec-01-2014 | ALL | EUR | Sportsbook | Won
    #convert language
    ${Sportsbook}    convert language    ${cf_language_file}    Products    Sportsbook
    ${Racing}    convert language    ${cf_language_file}    Products    Racing
    ${Number Game}    convert language    ${cf_language_file}    Products    Number Game
    ${Live Casino}    convert language    ${cf_language_file}    Products    Live Casino
    ${Virtual Sports}    convert language    ${cf_language_file}    Products    Virtual Sports
    ${BA}    convert language    ${cf_language_file}    Products    BA
    ${RNG Casino}    convert language    ${cf_language_file}    Products    RNG Casino
    ${Keno}    convert language    ${cf_language_file}    Products    Keno
    ${Microgaming}    convert language    ${cf_language_file}    Products    Microgaming
    ${AG Casino}    convert language    ${cf_language_file}    Products    AG Casino
    ${Gold Deluxe}    convert language    ${cf_language_file}    Products    Gold Deluxe
    ${Macau}    convert language    ${cf_language_file}    Products    Macau
    #get product code
    ${product_code}    Set Variable If    '${product}'=='${Sportsbook}' or '${product}'=='Sportsbook'    SB    '${product}'=='${Racing}' or '${product}'=='Racing'    RB    '${product}'=='${Number Game}' or '${product}'=='Number Game'
    ...    NG    '${product}'=='${Live Casino}' or '${product}'=='Live Casino'    LC    '${product}'=='${Virtual Sports}' or '${product}'=='Virtual Sports'    VS    '${product}'=='${BA}' or '${product}'=='BA'
    ...    BA    '${product}'=='${RNG Casino}' or '${product}'=='RNG Casino'    CS    '${product}'=='${Keno}' or '${product}'=='Keno'    KN    '${product}'=='${Microgaming}' or '${product}'=='Microgaming'
    ...    MGC    '${product}'=='${AG Casino}' or '${product}'=='AG Casino'    AGC    '${product}'=='${Gold Deluxe}' or '${product}'=='Gold Deluxe'    GD    '${product}'=='${Macau}' or '${product}'=='Macau'
    ...    VG
    #convert language
    ${currency}    Run Keyword If    '${currency}'=='ALL'    convert language    ${cf_language_file}    LRF    ${currency}
    ...    ELSE    Set Variable    ${currency}
    # select date
    select specific date    \    main    None    ${date}
    # select currency, based
    Select Window
    Select Frame    main
    Run Keyword If    '${currency}'!='None'    Select From List By Label    ${currency_listbox}    ${currency}
    Run Keyword If    '${based}'!='None'    Select From List By Label    ${based_listbox}    ${based}
    # select product
    Select From List By Value    ${product_listbox_daily_bet_list}    ${product_code}
    # select status
    Select From List By Value    ${status_listbox_daily_bet_list}    ${status}
    Unselect Frame
    # click Submit button
    click element on report    \    main    None    ${submit_button}
    wait page load

check bet list in daily bet list
    [Arguments]    ${date}    ${product}    ${excel file}    ${status}
    [Documentation]    Ex:
    ...
    ...    check bet list for daily bet list | date | Sportsbook | Lose
    # Navigate to REPORTS > Daily Bet List
    navigate left menu    REPORTS    Daily Bet List
    # filter product
    submit filters for daily bet list    ${date}    UUS    UUS    ${product}    ${status}
    # select 2000 items display in 1 page
    select listbox by value on report    \    main    None    xpath=//td/select[@class='ui-pg-selbox']    2000
    Sleep    3s
    Select Window
    Select Frame    main
    # Get Data from Excel File & Bet List. Then check them
    check bet list in daily bet list based on status    ${excel file}    ${product}    ${status}

check bet list in daily bet list based on status
    [Arguments]    ${excel file}    ${sheet}    ${status}
    [Documentation]    This action is used to check all tickets in the bet list.
    ...
    ...    Ex:
    ...
    ...    check bet list | Test_Data.xls | Sportsbook | Lose
    Open Excel Current Directory    ${excel file}
    Set Selenium Speed    0.1
    ${ac_totalrow_excel}    Get Row Count    ${sheet}
    : FOR    ${ac_index}    IN RANGE    1    ${ac_totalrow_excel}
    \    Comment    Get result in Excel File
    \    ${ac_result_excel}    Read Cell Data By Coordinates    ${sheet}    12    ${ac_index}
    \    ${ac_flag_check}    Run Keyword And Return Status    Should Be Equal    ${ac_result_excel}    ${status}
    \    Run Keyword If    '${ac_flag_check}'=='True'    check settled ticket data    ${sheet}    ${ac_index}
    Set Selenium Speed    ${cf_speed}
