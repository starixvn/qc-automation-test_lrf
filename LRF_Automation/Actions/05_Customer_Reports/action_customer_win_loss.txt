submit filters for customer win loss
    [Arguments]    ${from date}    ${to date}    ${currency}    ${based}    ${product}
    [Documentation]    This action is used to submit filters for Customer Win Loss report.
    ...
    ...    Example:
    ...
    ...    submit filters for customer win loss | Dec-01-2014 | Dec-03-2014 | ALL | EUR | Sportsbook, Racing
    # select date
    select date range    \    main    None    ${from date}    ${to date}
    # select currency, based
    Select Window
    Select Frame    main
    Run Keyword If    '${currency}'!='None'    Select From List By Label    ${currency_listbox}    ${currency}
    Run Keyword If    '${based}'!='None'    Select From List By Label    ${based_listbox}    ${based}
    # select products
    select products filter    \    main    None    ${product}
    Unselect Frame
    # click Submit button
    click element on report    \    main    None    ${submit_button}
    wait page load

check total customer wl in customer WL
    [Arguments]    ${window title}    ${main frame}    ${sub frame}    ${row number}
    [Documentation]    This action is used to check the first row of Total Customer WL according the below formular:
    ...    Total Customer W/L = -(Licensee /WL + BA Commission + Racing Revenue + Amt Due To BA + Amt Due To Racing)
    ...    It will show a message if they are not equal
    ...
    ...    Example:
    ...
    ...    check element on report | ${window title} | main | myframe
    ...
    ...    check element on report | ${window title} | main | None
    Run Keyword If    '${window title}'!='None'    Select Window    ${window title}
    Run Keyword If    '${main frame}'!='None'    Select Frame    ${main frame}
    Run Keyword If    '${sub frame}'!='None'    Select Frame    xpath=//iframe[contains(@src,'${sub frame}')]
    ${total_customer_wl}    Convert and remove comma from string to number    xpath=.//*[@id='data-content']/div[${row number}]/div[8]/div
    ${total_licensee_wl}    Convert and remove comma from string to number    xpath=.//*[@id='data-content']/div[${row number}]/div[10]
    ${ba_comission}    Convert and remove comma from string to number    xpath=.//*[@id='data-content']/div[${row number}]/div[11]
    ${racing_revenue}    Convert and remove comma from string to number    xpath=.//*[@id='data-content']/div[${row number}]/div[12]
    ${amt_due_to_ba}    Convert and remove comma from string to number    xpath=.//*[@id='data-content']/div[${row number}]/div[13]
    ${amt_due_to_racing}    Convert and remove comma from string to number    xpath=.//*[@id='data-content']/div[${row number}]/div[14]
    ${total_tmp}    Evaluate    -(${total_licensee_wl}+${ba_comission}+${racing_revenue}+${amt_due_to_ba}+${amt_due_to_racing})
    ${status}    Compare two number    ${total_customer_wl}    ${total_tmp}
    Run Keyword If    ${status}<>True    Log    Total customer W/L is not equal with the sum as the formular
    Unselect Frame
