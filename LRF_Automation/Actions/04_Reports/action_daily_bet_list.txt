submit filters for daily bet list
    [Arguments]    ${date}    ${currency}    ${based}    ${product}    ${status}
    [Documentation]    This action is used to submit filters for Daily Bet List report.
    ...
    ...    Example:
    ...
    ...    submit filters for daily bet list | Dec-01-2014 | ALL | EUR | Sportsbook | Won
    # select date
    select specific date    \    main    None    ${date}
    # select currency, based
    Select Window
    Select Frame    main
    Run Keyword If    '${currency}'!='None'    Select From List By Label    ${currency_listbox}    ${currency}
    Run Keyword If    '${based}'!='None'    Select From List By Label    ${based_listbox}    ${based}
    # select product
    Select From List By Label    ${product_listbox_daily_bet_list}    ${product}
    # select status
    Select From List By Label    ${status_listbox_daily_bet_list}    ${status}
    Unselect Frame
    # click Submit button
    click element on report    \    main    None    ${submit_button}
    wait page load

check bet list in daily bet list
    [Arguments]    ${date}    ${product}    ${excel file}
    [Documentation]    Ex:
    ...
    ...    check bet list for daily bet list | date | Sportsbook
    # filter product
    submit filters for daily bet list    ${date}    UUS    UUS    ${product}    Lose
    # select 2000 items display in 1 page
    select listbox by value on report    \    main    None    xpath=//td/select[@class='ui-pg-selbox']    2000
    Sleep    3s
    Select Window
    Select Frame    main
    # Get Data from Excel File & Bet List. Then check them
    check bet list for selected products    ${product}    ${excel file}
