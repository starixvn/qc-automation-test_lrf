navigate left menu

    ...    If Main Menu is already opened and you didn't want to click again, please input "None".
    ...
    ...    Example:

    ...

    # Click Main Menu label if needed
    Run Keyword If    '${mainmenu}'!='None'    Click Element    xpath=//span[contains(text(),'${mainmenu}')]
    # Click Sub Menu label
    # Handle special case: click duplicated labels "Handicap/Over Under/Live"
    ${ac_status}=    Evaluate    '${mainmenu}'=='FORECAST' and '${submenu}'=='Handicap/Over Under/Live'
    Run Keyword If    '${ac_status}'=='True'    Click Element    xpath=//td/a[contains(text(),'${submenu}') and contains(@href,'Forecast')]
    # Handle normal case
    Run Keyword If    '${ac_status}'!='True'    Click Element    xpath=//td/a[contains(text(),'${submenu}')]
    Unselect Frame

click number link in table
    [Arguments]    ${window title}    ${main frame}    ${sub frame}    ${table}    ${row}    ${from column}
    ...    ${to column}
    [Documentation]    This action will try to find the first number link inside a table. If the link is existed, we'll click it.
    ...
    ...    Example: Find if link existed in 2nd row, from 3rd column to 4th column. Then click it.
    ...
    ...    click number link in table | ${window title} | ${main frame} | ${sub frame} | ${report_table} | 2 | 3 | 4
    ...
    ...    click number link in table | ${window title} | ${main frame} | None | ${report_table} | 2 | 3 | 4
    Run Keyword If    '${window title}'!='None'    Select Window    ${window title}
    Run Keyword If    '${main frame}'!='None'    Select Frame    ${main frame}
    Run Keyword If    '${sub frame}'!='None'    Select Frame    xpath=//iframe[contains(@src,'${sub frame}')]
    # Find link in the range of columns. If existed, click on it
    : FOR    ${ac_index}    IN RANGE    ${from column}    ${to column}+1
    \    ${ac_numberlink_status}=    Run Keyword And Return Status    Page Should Contain Element    ${table}/tbody/tr[${row}]/td[${ac_index}]/a[contains(@class,'-number')]/span
    \    Run Keyword If    '${ac_numberlink_status}'=='True'    Click Element    ${table}/tbody/tr[${row}]/td[${ac_index}]/a[contains(@class,'-number')]/span
    \    Exit For Loop If    '${ac_numberlink_status}'=='True'
    \    Run Keyword If    '${ac_index}'=='${to column}' and '${ac_numberlink_status}'!='True'    Pass Execution    There is no link found in the table.
    Unselect Frame
