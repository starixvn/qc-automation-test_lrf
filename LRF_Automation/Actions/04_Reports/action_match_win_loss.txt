submit filters for match win loss
    [Arguments]    ${date}    ${licensee type}    ${currency}    ${based}    ${product}
    [Documentation]    This action is used to submit filters for Match Win Loss report.
    ...
    ...    ${product}= *'All'* or *'Sportsbook, Racing, Number Game, Live Casino, Virtual Sports, BA, RNG Casino, Keno'*
    ...
    ...    Example:
    ...
    ...    submit filters for match win loss | Dec-01-2014 | normal | ALL | EUR | Sportsbook, Keno
    # select date
    select specific date    \    main    None    ${date}
    # select type, currency, based
    Select Window
    Select Frame    main
    ${ac_type_status}=    Run Keyword And Return Status    Element Should Be Visible    ${type_listbox}
    Run Keyword If    '${licensee type}'!='None' and '${ac_type_status}'=='True'    Select From List By Label    ${type_listbox}    ${licensee type}
    Run Keyword If    '${currency}'!='None'    Select From List By Label    ${currency_listbox}    ${currency}
    Run Keyword If    '${based}'!='None'    Select From List By Label    ${based_listbox}    ${based}
    Unselect Frame
    # select product
    select products filter    \    main    None    ${product}
    # click Submit button
    click element on report    \    main    None    ${submit_button}
    wait page load

click match link of match win loss
    [Arguments]    ${window title}    ${main frame}    ${sub frame}    ${table}    ${row}    ${column}
    [Documentation]    This action will try to find the match link inside a table. If the link is existed, we'll click it.
    ...
    ...    Example: Find if link existed at 2nd row, 3rd column. Then click it.
    ...
    ...    click match link of match win loss | ${window title} | ${main frame} | ${sub frame} | ${report_table} | 2 | 3
    ...
    ...    click match link of match win loss | ${window title} | ${main frame} | None | ${report_table} | 2 | 3
    Run Keyword If    '${window title}'!='None'    Select Window    ${window title}
    Run Keyword If    '${main frame}'!='None'    Select Frame    ${main frame}
    Run Keyword If    '${sub frame}'!='None'    Select Frame    xpath=//iframe[contains(@src,'${sub frame}')]
    # Find link in the columns. If existed, click on it
    ${ac_matchlink_status}=    Run Keyword And Return Status    Page Should Contain Element    ${table}/tbody/tr[${row}]/td[${column}]/span[@class='match-link']
    Run Keyword If    '${ac_matchlink_status}'=='True'    Click Element    ${table}/tbody/tr[${row}]/td[${column}]/span[@class='match-link']
    Run Keyword If    '${ac_matchlink_status}'!='True'    Fail    No link found in the table. Please investigate the UI.
    Unselect Frame
