convert date format
    [Arguments]    ${input date}
    [Documentation]    This action is used to convert date from 'YYYY-MM-DD' format into 'MMM-DD-YYYY' format.
    ...
    ...    Example: Convert '2014-01-31' into 'Jan-31-2014'
    ...
    ...    ${variable}= | convert date format | 2014-01-31
    # Cut year, month, date from input date
    ${yyyy}=    Get Slice From List    ${input date}    0    4
    ${mm}=    Get Slice From List    ${input date}    5    7
    ${dd}=    Get Slice From List    ${input date}    8    10
    # Convert month from number into text
    ${mm}=    Set Variable If    '${mm}'=='01'    Jan    '${mm}'=='02'    Feb    '${mm}'=='03'
    ...    Mar    '${mm}'=='04'    Apr    '${mm}'=='05'    May    '${mm}'=='06'
    ...    Jun    '${mm}'=='07'    Jul    '${mm}'=='08'    Aug    '${mm}'=='09'
    ...    Sep    '${mm}'=='10'    Oct    '${mm}'=='11'    Nov    '${mm}'=='12'
    ...    Dec
    # Return converted date
    ${output date}=    Set Variable    ${mm}/${dd}/${yyyy}
    [Return]    ${output date}

select date range

    [Documentation]    This action is used to select date of from-to calendar.

    ...    select date range | ${window title} | ${main frame} | ${sub frame} | Dec-31-2014 | Jan-02-2015

    # Get month
    ${from_mm}=    Get Slice From List    ${from}    5    7
    ${from_mm}=    Convert To Integer    ${from_mm}
    ${from_data_month}=    Set Variable    ${from_mm - 1}
    ${to_mm}=    Get Slice From List    ${to}    5    7
    ${to_mm}=    Convert To Integer    ${to_mm}
    ${to_data_month}=    Set Variable    ${to_mm - 1}
    # split day-month-year of from date
    ${from}=    convert date format    ${from}
    ${dd1}=    Get Slice From List    ${from}    4    6
    ${dd11}=    Get Slice From List    ${dd1}    0    1
    ${dd12}=    Get Slice From List    ${dd1}    1    2
    Run Keyword If    '${dd11}'=='0'    Set Test Variable    ${dd1}    ${dd12}
    ${mm1}=    Get Slice From List    ${from}    0    3
    ${yyyy1}=    Get Slice From List    ${from}    7    11
    # split day-month-year of to date
    ${to}=    convert date format    ${to}
    ${dd2}=    Get Slice From List    ${to}    4    6
    ${dd21}=    Get Slice From List    ${dd2}    0    1
    ${dd22}=    Get Slice From List    ${dd2}    1    2
    Run Keyword If    '${dd21}'=='0'    Set Test Variable    ${dd2}    ${dd22}
    ${mm2}=    Get Slice From List    ${to}    0    3
    ${yyyy2}=    Get Slice From List    ${to}    7    11
    # click calendar picker to open calendar
    Click Element    ${from_calendar_picker}
    # select year-month in From calendar
    Wait Until Page Contains Element    ${year_listbox_of_from_calendar}
    Select From List By Value    ${year_listbox_of_from_calendar}    ${yyyy1}
    Select From List By Label    ${month_listbox_of_from_calendar}    ${mm1}
    # click day in From calendar
    Click Element    xpath=//*/div[1]/div[1]/table[1]/tbody/tr/td[@data-month='${from_data_month}']/a[text()='${dd1}'][contains(@class,'ui-state-default')]
    # select year-month in To calendar
    Select From List By Value    ${year_listbox_of_to_calendar}    ${yyyy2}
    Select From List By Label    ${month_listbox_of_to_calendar}    ${mm2}
    # click day in To calendar
    Click Element    xpath=//*/div[2]/div[1]/table[1]/tbody/tr/td[@data-month='${to_data_month}']/a[text()='${dd2}'][contains(@class,'ui-state-default')]
    # click OK button
    Click Element    ${ok_button_of_from_to_calendar}

wait for export file
    [Arguments]    ${filepath}
    [Documentation]    This action is used to wait for a export file.
    : FOR    ${ac_index}    IN RANGE    0    45
    \    ${ac_exist}    Run Keyword And Return Status    File Should Exist    ${filepath}
    \    ${ac_saving}    Run Keyword And Return Status    File Should Exist    ${filepath}.part
    \    Run Keyword If    '${ac_exist}'!='True' or '${ac_saving}'=='True'    sleep    2s
    \    Exit For Loop If    '${ac_index}'==45 or ('${ac_exist}'=='True' and '${ac_saving}'!='True')
