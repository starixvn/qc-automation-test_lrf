submit filters for top customer
    [Arguments]    ${currency}    ${based}    ${from date}    ${to date}    ${inactive since last}    ${system id}
    ...    ${sport}    ${min bet count}    ${sorting}    ${min total customer wl}    ${top}
    [Documentation]    This action is used to submit filters for Top Customer report.
    ...
    ...    ${sorting}='Best Win Loss' or 'Worst Win Loss'
    ...
    ...    Example:
    ...
    ...    submit filters for top customer | CNY | EUR | Dec-01-2014 | Dec-07-2014 | All | None | Soccer | 1 | Best Win Loss | 2 | 100
    ${currency}    Run Keyword If    '${currency}'=='ALL'    convert language    ${cf_language_file}    LRF    ${currency}
    ...    ELSE    Set Variable    ${currency}
    ${inactive since last}    Run Keyword If    '${inactive since last}'!='None'    convert language    ${cf_language_file}    LRF    ${inactive since last}
    ...    ELSE    Set Variable    ${inactive since last}
    ${sport}    Run Keyword If    '${sport}'!='None'    convert language    ${cf_language_file}    LRF    ${sport}
    ...    ELSE    Set Variable    ${sport}
    # select date
    select date range    \    main    None    ${from date}    ${to date}
    # select currency, based
    Select Window
    Select Frame    main
    Run Keyword If    '${currency}'!='None'    Select From List By Label    ${currency_listbox}    ${currency}
    Run Keyword If    '${based}'!='None'    Select From List By Label    ${based_listbox}    ${based}
    # select Inactive since last
    Run Keyword If    '${inactive since last}'!='None'    Select From List By Label    ${inactive_since_last_listbox}    ${inactive since last}
    # enter system ID
    Run Keyword If    '${system id}'!='None'    Input Text    ${system_id_textbox_top_customer}    ${system id}
    # select sport
    Run Keyword If    '${sport}'!='None'    Select From List By Label    ${sport_listbox_top_customer}    ${sport}
    # enter Min bet count - Min total customer WL
    Run Keyword If    '${min bet count}'!='None'    Input Text    ${min_bet_count_textbox}    ${min bet count}
    Run Keyword If    '${min total customer wl}'!='None'    Input Text    ${min_total_customer_wl_textbox}    ${min total customer wl}
    # select Top
    Run Keyword If    '${top}'!='None'    Select From List By Label    ${top_textbox}    ${top}
    # select Sorting
    Run Keyword If    '${sorting}'=='Best Win Loss'    Click Element    ${best_win_loss_radio_button}
    Run Keyword If    '${sorting}'=='Worst Win Loss'    Click Element    ${worst_win_loss_radio_button}
    # click Submit button
    click element on report    \    main    None    ${submit_button}
    wait page load
