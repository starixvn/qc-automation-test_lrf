submit filters for win loss

    ...    ${draw filter}= *'Normal'* or *'No Draws'* or *'No Draws For Tunrover Only'*

    Run Keyword If    '${based}'!='None'    Select From List By Label    ${based_listbox}    ${based}

    Run Keyword If    '${draw filter}'=='No Draws'    Click Element    ${no_draws_radio_button_win_loss}
    Run Keyword If    '${draw filter}'=='No Draws For Turnover Only'    Click Element    ${no_draws_for_turnover_radio_button_win_loss}
    # click Submit button
    click element on report    \    main    None    ${submit_button}
