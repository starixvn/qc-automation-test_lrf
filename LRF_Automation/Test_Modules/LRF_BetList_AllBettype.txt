TM_BL_ALL_01_Licensee Win Loss
    # Navigate to Licensee WL report
    navigate left menu    REPORTS    Licensee Win Loss
    # Submit filters for products
    submit filters for licensee win loss    ${cf_from_betlist}    ${cf_to_betlist}    UUS    Daily    Normal    No
    ...    ${cf_betlist_product}
    # Click UUS link to open customer list
    click element on report    \    main    None    xpath=//*[@id='1']/td/a[text()='UUS']
    wait page load
    # Click customer link to open bet list
    click element on report    \    main    None    xpath=//*[@id='1']/td/a[text()='${cf_betlist_user}']
    wait page load
    # Select to display 2000 tickets in one page
    select listbox by value on report    \    main    BetList    xpath=//td/select[@class='ui-pg-selbox']    2000
    Sleep    5s
    Select Window
    Select Frame    main
    Select Frame    xpath=//iframe[contains(@src,'BetList')]
    Set Selenium Speed    0.1
    # Get Data from Excel File & Bet List. Then check them
    check bet list for selected products    ${cf_betlist_product}    Data set\\Test_Data_All.xls
    Set Selenium Speed    ${cf_speed}
