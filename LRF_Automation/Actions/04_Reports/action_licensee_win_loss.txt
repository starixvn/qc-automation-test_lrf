submit filters for licensee win loss
    [Arguments]    ${from date}    ${to date}    ${currency}    ${report by}    ${draw filter}    ${exclude test data}
    ...    ${product}
    [Documentation]    This action is used to submit filters for Licensee Win Loss report and Win Loss By Products report.
    ...
    ...    ${report by}= *'Daily'* or *'Monthly'*
    ...
    ...    ${draw filter}= *'Normal'* or *'No Draws'* or *'No Draws For Tunrover Only'*
    ...
    ...    ${exclude test data}= *'Yes'* or *'No'*
    ...
    ...    ${product}= *'All'* or *'Sportsbook, Racing, Number Game, Live Casino, Virtual Sports, BA, RNG Casino, Keno'*
    ...
    ...    Example:
    ...
    ...    submit filters for licensee win loss | Dec-01-2014 | Dec-07-2014 | Daily | Normal | Yes | Sportsbook, Keno
    # select date
    select date range    \    main    None    ${from date}    ${to date}
    # select currency
    Select Window
    Select Frame    main
    Select From List By Label    ${currency_listbox}    ${currency}
    # select Report By
    Run Keyword If    '${report by}'=='Daily'    Click Element    ${daily_radio_button}
    Run Keyword If    '${report by}'=='Monthly'    Click Element    ${monthly_radio_button}
    # select Draw filter
    Run Keyword If    '${draw filter}'=='Normal'    Click Element    ${normal_radio_button}
    Run Keyword If    '${draw filter}'=='No Draws'    Click Element    ${no_draws_radio_button}
    Run Keyword If    '${draw filter}'=='No Draws For Turnover Only'    Click Element    ${no_draws_for_turnover_radio_button}
    # select Exclude Test Data
    Run Keyword If    '${exclude test data}'=='Yes'    Click Element    ${yes_radio_button_exclude_data}
    Run Keyword If    '${exclude test data}'=='No'    Click Element    ${no_radio_button_exclude_data}
    Unselect Frame
    # select products
    select products filter    \    main    None    ${product}
    # click Submit button
    click element on report    \    main    None    ${submit_button}
    wait page load

check report data for licensee win loss
    [Arguments]    ${from date}    ${to date}    ${product}
    [Documentation]    This action is used to check data of Licensee Win Loss report.
    ...
    ...    Example:
    ...
    ...    check report data for licensee win loss | Dec-01-2014 | Dec-07-2014 | Sportsbook
    # Get Data of Report from Excel file
    Open Excel Current Directory    ${cf_report_data_path}
    ${excel_turnover}=    Read Cell Data By Coordinates    ${product}    0    2
    ${excel_total_transaction}=    Read Cell Data By Coordinates    ${product}    1    2
    ${excel_customer_win_loss}=    Read Cell Data By Coordinates    ${product}    2    2
    ${excel_customer_comm}=    Read Cell Data By Coordinates    ${product}    3    2
    ${excel_customer_total}=    Read Cell Data By Coordinates    ${product}    4    2
    ${excel_licensee_win_loss}=    Read Cell Data By Coordinates    ${product}    5    2
    # Submit filters for products
    submit filters for licensee win loss    ${from date}    ${to date}    UUS    Monthly    Normal    No
    ...    ${product}
    # Click UUS link to open customer list
    click element on report    \    main    None    xpath=//*[@id='1']/td/a[text()='UUS']
    wait page load
    # Get data from web
    Select Window
    Select Frame    main
    ${ac_turnover}=    Get Text    xpath=//td[@title='iceEW']/parent::*/td[@aria-describedby='ReportGrid_TotalTurnOverSingle']
    ${ac_total_transaction}=    Get Text    xpath=//td[@title='iceEW']/parent::*/td[@aria-describedby='ReportGrid_TotalTransaction']
    ${ac_customer_win_loss}=    Get Text    xpath=//td[@title='iceEW']/parent::*/td[@aria-describedby='ReportGrid_CustomerWinLoss']
    ${ac_customer_comm}=    Get Text    xpath=//td[@title='iceEW']/parent::*/td[@aria-describedby='ReportGrid_CustomerComm']
    ${ac_customer_total}=    Get Text    xpath=//td[@title='iceEW']/parent::*/td[@aria-describedby='ReportGrid_CustomerTotal']
    ${ac_licensee_win_loss}=    Get Text    xpath=//td[@title='iceEW']/parent::*/td[@aria-describedby='ReportGrid_TotalLicenseeWLSingle']
    # Check data of report
    Should Be Equal    ${ac_turnover}    ${excel_turnover}
    Should Be Equal    ${ac_total_transaction}    ${excel_total_transaction}
    Should Be Equal    ${ac_customer_win_loss}    ${excel_customer_win_loss}
    Should Be Equal    ${ac_customer_comm}    ${excel_customer_comm}
    Should Be Equal    ${ac_customer_total}    ${excel_customer_total}
    Should Be Equal    ${ac_licensee_win_loss}    ${excel_licensee_win_loss}
