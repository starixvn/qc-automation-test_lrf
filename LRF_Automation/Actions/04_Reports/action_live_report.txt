submit filters for live report
    [Arguments]    ${from date}    ${to date}    ${currency}    ${based}    ${live status}
    [Documentation]    This action is used to submit filters for Live Report report.
    ...
    ...    Example:
    ...
    ...    submit filters for live report | Dec-01-2014 | Dec-03-2014 | \ ALL | EUR | Live
    # select date
    select date range    \    main    None    ${from date}    ${to date}
    # select currency, based
    Select Window
    Select Frame    main
    Run Keyword If    '${currency}'!='None'    Select From List By Label    ${currency_listbox}    ${currency}
    Run Keyword If    '${based}'!='None'    Select From List By Label    ${based_listbox}    ${based}
    # select live status
    Select From List By Label    ${live_status_listbox}    ${live status}
    Unselect Frame
    # click Submit button
    click element on report    \    main    None    ${submit_button_live_report}
    wait page load

check export excel function for live report
    [Arguments]    ${window title}    ${main frame}    ${sub frame}    ${checked file name}
    [Documentation]    This action is used to check Export To Excel function.
    ...
    ...    Example: Check file 'RunningBetList.xls' existed after exporting sucessfully. And exporting file need 5 seconds to be saved.
    ...
    ...    check export excel function | ${window title} | ${main frame} | ${sub frame} | 5s | RunningBetList.xls
    Run Keyword If    '${window title}'!='None'    Select Window    ${window title}
    Run Keyword If    '${main frame}'!='None'    Select Frame    ${main frame}
    Run Keyword If    '${sub frame}'!='None'    Select Frame    xpath=//iframe[contains(@src,'${sub frame}')]
    # Check to see Folder existed or not. If not, we'll create
    create test directory    ${cf_export_dir}
    # Click Export
    Click Element    ${export_to_excel_button_live_report}
    # Check file existed
    wait for export file    ${cf_export_dir}${checked file name}
    File Should Exist    ${cf_export_dir}${checked file name}
    Empty Directory    ${cf_export_dir}
    Unselect Frame
