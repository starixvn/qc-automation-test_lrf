submit filters for win loss by products
    [Arguments]    ${from date}    ${to date}    ${currency}    ${report by}    ${draw filter}    ${exclude test data}
    ...    ${product}
    [Documentation]    This action is used to submit filters for Win Loss By Products report
    ...
    ...    ${report by}= *'Daily'* or *'Monthly'*
    ...
    ...    ${draw filter}= *'Normal'* or *'No Draws'* or *'No Draws For Tunrover Only'*
    ...
    ...    ${exclude test data}= *'Yes'* or *'No'*
    ...
    ...    ${product}= *'All'* or *'Sportsbook, Racing, Number Game, Live Casino, Virtual Sports, BA, RNG Casino, Keno'*
    ...
    ...    Example:
    ...
    ...    submit filters for win loss by products | Dec-01-2014 | Dec-07-2014 | Daily | Normal | Yes | Sportsbook, Keno
    # select date
    select date range    \    main    None    ${from date}    ${to date}
    # select currency
    Select Window
    Select Frame    main
    Select From List By Label    ${currency_listbox}    ${currency}
    # select Report By
    Run Keyword If    '${report by}'=='Daily'    Click Element    ${daily_radio_button_wl_by_products}
    Run Keyword If    '${report by}'=='Monthly'    Click Element    ${monthly_radio_button_wl_by_products}
    # select Draw filter
    Run Keyword If    '${draw filter}'=='Normal'    Click Element    ${normal_radio_button_wl_by_products}
    Run Keyword If    '${draw filter}'=='No Draws'    Click Element    ${no_draws_radio_button_wl_by_products}
    Run Keyword If    '${draw filter}'=='No Draws For Turnover Only'    Click Element    ${no_draws_for_turnover_radio_button_wl_by_products}
    # select Exclude Test Data
    Run Keyword If    '${exclude test data}'=='Yes'    Click Element    ${yes_radio_button_wl_by_products}
    Run Keyword If    '${exclude test data}'=='No'    Click Element    ${no_radio_button_wl_by_products}
    Unselect Frame
    # select products
    select products filter    \    main    None    ${product}
    # click Submit button
    click element on report    \    main    None    ${submit_button}
    wait page load

check customer wl in WL by Products
    [Arguments]    ${window title}    ${main frame}    ${sub frame}    ${row_number}
    [Documentation]    This action is used to check the value of total customer wl and licensee wl. we have the difference formular to calculate accroding the products. For Racing and BA products:
    ...    Total customer wl = -(Comm + Amount Due)
    ...    For the others product, the formular is:
    ...    Total customer wl = -(Licensee WL)
    ...    It will show a message if they are not equal
    Run Keyword If    '${window title}'!='None'    Select Window    ${window title}
    Run Keyword If    '${main frame}'!='None'    Select Frame    ${main frame}
    Run Keyword If    '${sub frame}'!='None'    Select Frame    xpath=//iframe[contains(@src,'${sub frame}')]
    ${total_customer_wl}    Convert and remove comma from string to number    xpath=.//*[@id='${row_number}']/td[11]
    ${total_licensee_wl}    Convert and remove comma from string to number    xpath=.//*[@id='${row_number}']/td[12]
    ${total_licensee_wl}    Evaluate    ${total_licensee_wl}*-1
    ${product}    Get Text    xpath=.//*[@id='${row_number}']/td[6]
    ${status}    Run Keyword If    '${product}'=='Racing' or '${product}'=='BA'    check total customer wl of racing product    ${row_number}
    ...    ELSE    Compare two number    ${total_customer_wl}    ${total_licensee_wl}
    Run Keyword If    ${status}<>True    Log    They are not equal
    Unselect Frame

check total customer wl of racing product
    [Arguments]    ${row_number}
    [Documentation]    This action is used to check Total customer of \ Racing and BA product.
    ${comm}    Convert and remove comma from string to number    xpath=.//*[@id='${row_number}']/td[13]
    ${amount_due}    Convert and remove comma from string to number    xpath=.//*[@id='${row_number}']/td[14]
    ${total_customer_wl}    Convert and remove comma from string to number    xpath=.//*[@id='${row_number}']/td[11]
    ${tmp}    Evaluate    -(${comm}+${amount_due})
    ${status}    Compare two number    ${tmp}    ${total_customer_wl}
    [Return]    ${status}

loop for checking customer wl by product
    Select frame    main
    ${i}    Set Variable    1
    ${row_number}    Get Matching Xpath Count    xpath=.//*[@id='ReportGrid']//tr[@tabindex="-1"]
    : FOR    ${i}    IN RANGE    1    ${row_number}
    \    check customer wl in WL by Products    \    main    None    ${i}
    \    ${i}    Evaluate    ${i} + 1
    \    Exit For Loop If    '${i}' == '${row_number}'
    Unselect Frame
